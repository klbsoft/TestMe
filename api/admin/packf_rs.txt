pub async fn list_users() {
    println!("\n=== ALL USERS ===\n");
    
    let mut cursor = match db().users().find(doc! {}).await {
        Ok(cursor) => cursor,
        Err(e) => {
            println!("Error: {}", e);
            return;
        }
    };
    
    let mut count = 0;
    while cursor.advance().await.unwrap_or(false) {
        match cursor.deserialize_current() {
            Ok(user) => {
                count += 1;
                println!("User {}: {:#?}", count, user);
            }
            Err(e) => {
                println!("Error deserializing: {}", e);
            }
        }
    }
    println!("Total: {}", count);
}

use sha2::{Digest, Sha512};

/// Stage 1: deterministic SHA-512, byte-identical to the JS tool.
pub fn deterministic_hash_512(input: &str) -> String {
    let mut hasher = Sha512::new();
    hasher.update(input.as_bytes());
    hex::encode(hasher.finalize())
}

/// Stage 2: non-deterministic re-hash, mirrors `CryptoService::non_deterministic_hash`.
/// Returns (hash_string, salt_string).
pub fn non_deterministic_hash(prehash_hex: &str) -> anyhow::Result<(String, String)> {
    use argon2::{Argon2, PasswordHasher};
    use argon2::password_hash::{rand_core::OsRng, SaltString};

    let salt = SaltString::generate(&mut OsRng);
    let salt_string = salt.as_str().to_string();

    let hash = Argon2::default()
        .hash_password(prehash_hex.as_bytes(), &salt)?
        .to_string();

    Ok((hash, salt_string))
}

use argon2::password_hash::rand_core::{OsRng, RngCore};

/// 64-char hex id (32 random bytes).
pub fn random_id_64() -> String {
    let mut bytes = [0u8; 32];
    OsRng.fill_bytes(&mut bytes);
    hex::encode(bytes)
}

use anyhow::{Context, Result};
use clap::Parser;
use std::io::{self, Write};

use components::dbfactory::{db, init_db};
use identity::identities::Teacher;

#[derive(Parser, Debug)]
#[command(name = "teacher-cli", about = "Create a teacher account")]
struct Args {
    #[arg(short, long)]
    name: String,

    /// Stage-1 SHA-512 hex (128 chars) from the JS tool.
    #[arg(short = 'p', long = "password-hash", conflicts_with = "raw_password")]
    password_hash: Option<String>,

    /// Raw password; stage 1 runs here.
    #[arg(long = "raw-password")]
    raw_password: Option<String>,

    #[arg(long)]
    dry_run: bool,
}

#[tokio::main]
async fn main() -> Result<()> {
    let args = Args::parse();

    // ── STAGE 1 (deterministic) ─────────────────────────────────────────
    let prehash_hex = match (args.password_hash, args.raw_password) {
        (Some(h), None) => {
            let h = h.trim().to_lowercase();
            if h.len() != 128 || !h.chars().all(|c| c.is_ascii_hexdigit()) {
                anyhow::bail!("password-hash must be 128 hex chars (SHA-512)");
            }
            h
        }
        (None, Some(raw)) => hash::deterministic_hash_512(&raw),
        (None, None) => {
            print!("Password (visible): ");
            io::stdout().flush()?;
            let mut buf = String::new();
            io::stdin().read_line(&mut buf)?;
            hash::deterministic_hash_512(buf.trim_end_matches(['\n', '\r']))
        }
        _ => unreachable!("clap conflicts_with"),
    };

    // ── STAGE 2 (non-deterministic) ─────────────────────────────────────
    let (password_hash, salt) =
        hash::non_deterministic_hash(&prehash_hex).context("non-deterministic re-hash")?;

    // ── Build Teacher (exact shape your factory serializes) ─────────────
    let now = chrono::Utc::now().to_rfc3339();
    let teacher = Teacher {
        id: id::random_id_64(),
        name: args.name.clone(),
        password_hash,
        salt,
        created_at: now.clone(),
        updated_at: now,
    };
    let teacher_json = serde_json::to_value(&teacher)?;

    if args.dry_run {
        println!("{}", serde_json::to_string_pretty(&teacher_json)?);
        return Ok(());
    }

    // ── Initialize your factory, then insert via .teachers() ────────────
    init_db().await;

    db().teachers()
        .insert_one(&teacher)
        .await
        .context("insert teacher")?;

    println!("{}", serde_json::to_string(&teacher_json)?);
    Ok(())
}
