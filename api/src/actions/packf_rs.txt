use crate::components::dbfactory::db;
use mongodb::bson::doc;

/// Delete every session whose `ends_at` is in the past.
/// One `delete_many` — no per-document loop.
pub async fn purge_expired_sessions() {
    let now = chrono::Utc::now().to_rfc3339();

    match db()
        .sessions()
        .delete_many(doc! { "ends_at": { "$lt": &now } })
        .await
    {
        Ok(res) => {
            if res.deleted_count > 0 {
                println!("[cleanup] purged {} expired session(s)", res.deleted_count);
            }
        }
        Err(e) => {
            println!("[cleanup] PURGE SESSIONS ERROR: {}", e);
        }
    }
}

/// Spawn a background task that purges expired sessions every 30 minutes.
/// Call once at startup, right after `init_db().await`.
pub fn spawn_session_cleanup() {
    tokio::spawn(async {
        // run once immediately so a fresh boot also cleans up
        purge_expired_sessions().await;

        let mut interval = 
        tokio::time::interval(std::time::Duration::from_secs(30 * 60));
        interval.tick().await; // consume the immediate first tick

        loop {
            interval.tick().await;
            purge_expired_sessions().await;
        }
    });
}

pub async fn complete_attempt(
    Json(body): Json<AssigmentAttemptResource>,
) -> impl IntoResponse {
    println!("════════════════════════════════════════════════════════════");
    println!("[COMPLETE_ATTEMPT] ── INCOMING REQUEST ───────────────────");

    // ── 1) LOAD THE SESSION ─────────────────────────────────────────
    println!("[COMPLETE_ATTEMPT] STEP 1 ── looking up session");
    let session: Session = match db()
        .sessions()
        .find_one(doc! { "token": &body.token })
        .await
    {
        Ok(Some(s)) => {
            println!(
                "[COMPLETE_ATTEMPT] STEP 1 ✅ session found (id = {:?}, user = {:?})",
                s.id, s.user
            );
            s
        }
        Ok(None) => {
            println!("[COMPLETE_ATTEMPT] STEP 1 ❌ no session for that token");
            return "FAIL:Invalid session".to_string();
        }
        Err(e) => {
            println!("[COMPLETE_ATTEMPT] STEP 1 ❌ DB error: {}", e);
            return "FAIL:Database error".to_string();
        }
    };

    // ── 2) SESSION NOT EXPIRED ──────────────────────────────────────
    println!("[COMPLETE_ATTEMPT] STEP 2 ── checking expiry");
    let now = chrono::Utc::now();
    let ends_at = match chrono::DateTime::parse_from_rfc3339(&session.ends_at) {
        Ok(dt) => dt.with_timezone(&chrono::Utc),
        Err(e) => {
            println!("[COMPLETE_ATTEMPT] STEP 2 ❌ parse error: {}", e);
            return "FAIL:Invalid session format".to_string();
        }
    };
    if now >= ends_at {
        println!("[COMPLETE_ATTEMPT] STEP 2 ❌ session expired");
        return "FAIL:Session expired".to_string();
    }
    println!("[COMPLETE_ATTEMPT] STEP 2 ✅ session not expired");

    // ── 3) FIND THE ATTEMPT BY session.id ───────────────────────────
    // session.id == attempt.id (set at creation in start_attempt).
    println!(
        "[COMPLETE_ATTEMPT] STEP 3 ── looking up attempt by id = {:?}",
        session.id
    );
    let attempt: AssigmentAttempt = match db()
        .attempts()
        .find_one(doc! { "id": &session.id })
        .await
    {
        Ok(Some(a)) => {
            println!(
                "[COMPLETE_ATTEMPT] STEP 3 ✅ attempt found (user = {:?}, assigment_id = {:?})",
                a.user, a.assigment_id
            );
            a
        }
        Ok(None) => {
            println!("[COMPLETE_ATTEMPT] STEP 3 ❌ no attempt for session.id = {:?}", session.id);
            return "FAIL:Attempt not found".to_string();
        }
        Err(e) => {
            println!("[COMPLETE_ATTEMPT] STEP 3 ❌ DB error: {}", e);
            return "FAIL:Database error".to_string();
        }
    };

    // ── 4) SIZE CAP ─────────────────────────────────────────────────
    if body.json.len() > 10_000_000 {
        println!(
            "[COMPLETE_ATTEMPT] STEP 4 ❌ payload too large ({} bytes)",
            body.json.len()
        );
        return "FAIL:Payload too large (max 10 MB)".to_string();
    }
    println!("[COMPLETE_ATTEMPT] STEP 4 ✅ payload {} bytes", body.json.len());

    // ── 5) FILL THE ATTEMPT WITH THE ANSWER JSON ────────────────────
    println!("[COMPLETE_ATTEMPT] STEP 5 ── writing answers to attempt");
    match db()
        .attempts()
        .update_one(
            doc! { "id": &attempt.id },
            doc! { "$set": {
                "json": &body.json,
            }},
        )
        .await
    {
        Ok(res) if res.matched_count > 0 => {
            println!(
                "[COMPLETE_ATTEMPT] STEP 5 ✅ attempt updated ({} bytes written)",
                body.json.len()
            );
        }
        Ok(_) => {
            println!("[COMPLETE_ATTEMPT] STEP 5 ❌ attempt vanished mid-update");
            return "FAIL:Attempt not found".to_string();
        }
        Err(e) => {
            println!("[COMPLETE_ATTEMPT] STEP 5 ❌ DB error: {}", e);
            return "FAIL:Database error".to_string();
        }
    }

    // ── 6) DELETE THE SESSION ───────────────────────────────────────
    println!("[COMPLETE_ATTEMPT] STEP 6 ── deleting session");
    match db()
        .sessions()
        .delete_one(doc! { "token": &body.token })
        .await
    {
        Ok(_) => println!("[COMPLETE_ATTEMPT] STEP 6 ✅ session deleted"),
        Err(e) => {
            // non-fatal — the session will expire on its own
            println!("[COMPLETE_ATTEMPT] STEP 6 ⚠ delete error: {}", e);
        }
    }

    // ── 7) RETURN ───────────────────────────────────────────────────
    println!("[COMPLETE_ATTEMPT] STEP 7 ✅ ── DONE ──");
    println!("════════════════════════════════════════════════════════════");
    format!("OK:{}", attempt.id)
}

            }
            Err(e) => {
                println!("[ASSIGMENT_INFO] STEP 5 ⚠ attempt lookup error: {}", e);
            }
        }
    } else {
        println!("[ASSIGMENT_INFO] STEP 5   caller is a teacher — no per-attempt grade");
    }

    // ── 6) RETURN ───────────────────────────────────────────────────
    let response = AssigmentInfo {
        label: assignment.label,
        assigment_type: assignment.assigment_type,
        grade,
    };

    match serde_json::to_string(&response) {
        Ok(json) => {
            println!("[ASSIGMENT_INFO] STEP 6 ✅ ── DONE ── {}", json);

            println!("════════════════════════════════════════════════════════════");
            "FAIL:Failed to serialize result".to_string()
        }
    }
}

use axum::{response::IntoResponse, Json};
use mongodb::bson::doc;
use serde::Serialize;

use crate::components::dbfactory::db;
use crate::identity::identities::{
    AssigmentAttempt, AttemptRow, ResourceAccess, Session, Teacher, UserInfo,
};

 

pub async fn get_attempts(Json(body): Json<ResourceAccess>) -> impl IntoResponse {
    // ── 1) SESSION ──────────────────────────────────────────────────
    let session: Session = match db()
        .sessions()
        .find_one(doc! { "token": &body.token })
        .await
    {
        Ok(Some(s)) => s,
        Ok(None) => return "FAIL:Invalid session".to_string(),
        Err(e) => {
            println!("GET ATTEMPTS SESSION ERROR: {}", e);
            return "FAIL:Database error".to_string();
        }
    };

    // ── 2) EXPIRY ───────────────────────────────────────────────────
    let now = chrono::Utc::now();
    let ends_at = match chrono::DateTime::parse_from_rfc3339(&session.ends_at) {
        Ok(dt) => dt.with_timezone(&chrono::Utc),
        Err(e) => {
            println!("GET ATTEMPTS PARSE ERROR: {}", e);
            return "FAIL:Invalid session format".to_string();
        }
    };
    if now >= ends_at {
        return "FAIL:Session expired".to_string();
    }

    // ── 3) RESOLVE ROLE ─────────────────────────────────────────────
    let teacher_opt: Option<Teacher> = db()
        .teachers()
        .find_one(doc! { "id": &session.user })
        .await
        .ok()
        .flatten();

    if let Some(teacher) = teacher_opt {
        // ── TEACHER PATH: all attempts from all their students ──────
        let mut students_by_id: std::collections::HashMap<String, (String, String)> =
            std::collections::HashMap::new();

        let mut students_cursor = match db()
            .users()
            .find(doc! { "teacher": &teacher.id })
            .await
        {
            Ok(c) => c,
            Err(e) => {
                println!("GET ATTEMPTS STUDENTS ERROR: {}", e);
                return "FAIL:Database error".to_string();
            }
        };

        while students_cursor.advance().await.unwrap_or(false) {
            if let Ok(u) = students_cursor.deserialize_current() {
                students_by_id.insert(u.id.clone(), (u.code, u.group));
            }
        }

        if students_by_id.is_empty() {
            return "OK:[]".to_string();
        }

        let student_ids: Vec<String> = students_by_id.keys().cloned().collect();

        let mut cursor = match db()
            .attempts()
            .find(doc! { "user": { "$in": &student_ids } })
            .await
        {
            Ok(c) => c,
            Err(e) => {
                println!("GET ATTEMPTS FIND ERROR: {}", e);
                return "FAIL:Database error".to_string();
            }
        };

        let mut rows: Vec<AttemptRow> = Vec::new();

        while cursor.advance().await.unwrap_or(false) {
            let a: AssigmentAttempt = match cursor.deserialize_current() {
                Ok(a) => a,
                Err(e) => {
                    println!("GET ATTEMPTS DESERIALIZE ERROR: {}", e);
                    continue;
                }
            };

            let Some((code, group)) = students_by_id.get(&a.user) else {
                continue;
            };

            rows.push(AttemptRow {
                id: a.id,
                user: a.user,
                code: code.clone(),
                group: group.clone(),
                attempt: a.attempt,
                assigment_label: a.assigment_label,
                assigment_type: a.assigment_type,
                assigment_id: a.assigment_id,
                max_attempts: a.max_attempts,
                grade: a.grade,
                window: a.window,
                json: a.json,
            });
        }

        rows.sort_by(|a, b| {
            a.group
                .cmp(&b.group)
                .then_with(|| numeric_suffix(&a.code).cmp(&numeric_suffix(&b.code)))
                .then_with(|| a.attempt.cmp(&b.attempt))
        });

        return match serde_json::to_string(&rows) {
            Ok(json) => format!("OK:{}", json),
            Err(e) => {
                println!("GET ATTEMPTS SERIALIZE ERROR: {}", e);
                "FAIL:Failed to serialize result".to_string()
            }
        };
    }

    // ── STUDENT PATH: all attempts belonging to this student ────────
    let student: UserInfo = match db()
        .users()
        .find_one(doc! { "id": &session.user })
        .await
    {
        Ok(Some(s)) => s,
        Ok(None) => return "FAIL:User not found".to_string(),
        Err(e) => {
            println!("GET ATTEMPTS USER ERROR: {}", e);
            return "FAIL:Database error".to_string();
        }
    };

    let mut cursor = match db()
        .attempts()
        .find(doc! { "user": &student.id })
        .await
    {
        Ok(c) => c,
        Err(e) => {
            println!("GET ATTEMPTS FIND ERROR: {}", e);
            return "FAIL:Database error".to_string();
        }
    };

    let mut rows: Vec<AttemptRow> = Vec::new();

    while cursor.advance().await.unwrap_or(false) {
        let a: AssigmentAttempt = match cursor.deserialize_current() {
            Ok(a) => a,
            Err(e) => {
                println!("GET ATTEMPTS DESERIALIZE ERROR: {}", e);
                continue;
            }
        };

        rows.push(AttemptRow {
            id: a.id,
            user: a.user,
            code: student.code.clone(),
            group: student.group.clone(),
            attempt: a.attempt,
            assigment_label: a.assigment_label,
            assigment_type: a.assigment_type,
            assigment_id: a.assigment_id,
            max_attempts: a.max_attempts,
            grade: a.grade,
            window: a.window,
            json: String::new(),   // student sees no submission json
        });
    }

    rows.sort_by(|a, b| a.attempt.cmp(&b.attempt));

    match serde_json::to_string(&rows) {
        Ok(json) => format!("OK:{}", json),
        Err(e) => {
            println!("GET ATTEMPTS SERIALIZE ERROR: {}", e);
            "FAIL:Failed to serialize result".to_string()
        }
    }
}

fn numeric_suffix(code: &str) -> u32 {
    code.chars()
        .skip_while(|c| !c.is_ascii_digit())
        .collect::<String>()
        .parse()
        .unwrap_or(0)
}

    extract::{
        ws::{Message, WebSocket, WebSocketUpgrade},
        Query,
    },
    response::IntoResponse,

use mongodb::bson::doc;

// ─────────────────────────────────────────────────────────────────────
// Tunables — tweak here

/// How often the server sends a Ping frame.
const HEARTBEAT_PING_INTERVAL_SECS: u64 = 1;

/// How long the server tolerates silence from the client before
/// considering the socket dead and closing it. Must be greater than
/// the client's pong latency. Also acts as the "no pong received" cap
/// even if the TCP socket is still technically open.
const HEARTBEAT_TIMEOUT_SECS: u64 = 90;

// ─────────────────────────────────────────────────────────────────────

#[derive(Debug, Deserialize)]
pub struct HeartbeatQuery {
    pub token: String,
}

/// GET /heartbeat?token=...
pub async fn get_heartbeat(
    ws: WebSocketUpgrade,
    Query(q): Query<HeartbeatQuery>,
) -> impl IntoResponse {
    println!("════════════════════════════════════════════════════════════");
    println!("[HEARTBEAT] ── INCOMING UPGRADE ───────────────────────────");
    println!("[HEARTBEAT] token = {:?}", mask(&q.token));
    ws.on_upgrade(move |socket| handle(socket, q.token))
}

async fn handle(mut socket: WebSocket, token: String) {
    let connected_at = Instant::now();
    println!("[HEARTBEAT] STEP 1 ── socket accepted, looking up session");
    println!("[HEARTBEAT] STEP 1   token = {:?}", mask(&token));

    let session: Session = match db()
        .sessions()
        .find_one(doc! { "token": &token })
        .await
    {
        Ok(Some(s)) => {
            println!(
                "[HEARTBEAT] STEP 1 ✅ session found (id = {:?}, user = {:?})",
                s.id, s.user
            );
            s
        }
        Ok(None) => {
            println!("[HEARTBEAT] STEP 1 ❌ no session for that token");
            let _ = socket
                .send(Message::Text("FAIL:Invalid session".into()))
                .await;
            let _ = socket.close().await;
            return;
        }
        Err(e) => {
            println!("[HEARTBEAT] STEP 1 ❌ DB error: {}", e);
            let _ = socket
                .send(Message::Text("FAIL:Database error".into()))
                .await;
            let _ = socket.close().await;
            return;
        }
    };

    println!("[HEARTBEAT] STEP 2 ── checking expiry");
    let now = chrono::Utc::now();
    let ends_at = match chrono::DateTime::parse_from_rfc3339(&session.ends_at) {
        Ok(dt) => dt.with_timezone(&chrono::Utc),
        Err(e) => {
            println!("[HEARTBEAT] STEP 2 ❌ parse error: {}", e);
            let _ = socket
                .send(Message::Text("FAIL:Invalid session format".into()))
                .await;
            let _ = socket.close().await;
            return;
        }
    };
    println!("[HEARTBEAT] STEP 2   now = {}, ends_at = {}", now, ends_at);
    if now >= ends_at {
        println!("[HEARTBEAT] STEP 2 ❌ expired");
        let _ = socket
            .send(Message::Text("FAIL:Session expired".into()))
            .await;
        let _ = socket.close().await;
        return;
    }
    println!("[HEARTBEAT] STEP 2 ✅ session not expired");

    println!(
        "[HEARTBEAT] STEP 3 ✅ connected (user = {:?}, expires = {})",
        session.user, session.ends_at
    );

    if let Err(e) = socket.send(Message::Text("OK:connected".into())).await {
        println!("[HEARTBEAT] STEP 3 ⚠ failed to send OK:connected: {}", e);
        return;
    }
    println!("[HEARTBEAT] STEP 3 ✅ sent OK:connected to client");

    println!(
        "[HEARTBEAT] STEP 4 ── entering keep-alive loop (ping every {} s, timeout {} s)",
        HEARTBEAT_PING_INTERVAL_SECS, HEARTBEAT_TIMEOUT_SECS
    );

    let mut tick = interval(Duration::from_secs(HEARTBEAT_PING_INTERVAL_SECS));
    let mut tick_count: u64 = 0;

    // last time we received *any* frame from the client
    let mut last_seen = Instant::now();
    // last time we sent a ping that hasn't been pong'd
    let mut last_ping_sent: Option<Instant> = None;

    loop {
        tokio::select! {
            _ = tick.tick() => {
                tick_count += 1;
                println!(
                    "[HEARTBEAT] STEP 4   tick #{} — sending ping (last client frame {} s ago)",
                    tick_count,
                    last_seen.elapsed().as_secs()
                );
                if let Err(e) = socket.send(Message::Ping(Vec::new())).await {
                    println!("[HEARTBEAT] STEP 4 ❌ ping failed (client gone): {}", e);
                    break;
                }
                last_ping_sent = Some(Instant::now());
                println!("[HEARTBEAT] STEP 4   tick #{} ✅ ping sent", tick_count);

                // timeout check: if we sent a ping and never heard back
                // within HEARTBEAT_TIMEOUT_SECS, the connection is dead
                // even if the TCP layer hasn't noticed yet.
                if let Some(sent_at) = last_ping_sent {
                    if sent_at.elapsed().as_secs() >= HEARTBEAT_TIMEOUT_SECS {
                        println!(
                            "[HEARTBEAT] STEP 4 ❌ no pong for {} s (last client frame {} s ago) — closing",
                            sent_at.elapsed().as_secs(),
                            last_seen.elapsed().as_secs()
                        );
                        let _ = socket.send(Message::Close(None)).await;
                        break;
                    }
                }
            }

            msg = socket.recv() => {
                match msg {
                    Some(Ok(Message::Text(t))) => {
                        last_seen = Instant::now();
                        last_ping_sent = None; // any client frame clears the pending ping
                        println!(
                            "[HEARTBEAT] STEP 4   ← text frame (len = {}) = {:?} (last client frame reset)",
                            t.len(),
                            if t.len() > 64 { &t[..64] } else { &t }
                        );
                        if t == "ping" {
                            println!("[HEARTBEAT] STEP 4   ← client ping, replying pong");
                            if let Err(e) = socket.send(Message::Text("pong".into())).await {
                                println!("[HEARTBEAT] STEP 4 ❌ pong send failed: {}", e);
                                break;
                            }
                            println!("[HEARTBEAT] STEP 4   ✅ pong sent");
                        }
                    }
                    Some(Ok(Message::Binary(b))) => {
                        last_seen = Instant::now();
                        last_ping_sent = None;
                        println!(
                            "[HEARTBEAT] STEP 4   ← binary frame ({} bytes), ignoring",
                            b.len()
                        );
                    }
                    Some(Ok(Message::Pong(p))) => {
                        last_seen = Instant::now();
                        last_ping_sent = None;
                        println!(
                            "[HEARTBEAT] STEP 4   ← pong ({} bytes) — connection alive (ping→pong {} s)",
                            p.len(),
                            last_ping_sent.map(|t| t.elapsed().as_secs()).unwrap_or(0)
                        );
                    }
                    Some(Ok(Message::Ping(p))) => {
                        last_seen = Instant::now();
                        last_ping_sent = None;
                        println!(
                            "[HEARTBEAT] STEP 4   ← ping from client ({} bytes) — axum auto-replies",
                            p.len()
                        );
                    }
                    Some(Ok(Message::Close(frame))) => {
                        let lifetime = connected_at.elapsed().as_secs();
                        match frame {
                            Some(f) => println!(
                                "[HEARTBEAT] STEP 4   ← close frame (code = {}, reason = {:?}) — socket lived {} s",
                                f.code, f.reason, lifetime
                            ),
                            None => println!(
                                "[HEARTBEAT] STEP 4   ← close frame (no body) — socket lived {} s",
                                lifetime
                            ),
                        }
                        break;
                    }
                    None => {
                        println!(
                            "[HEARTBEAT] STEP 4   ← stream ended (None) — socket lived {} s",
                            connected_at.elapsed().as_secs()
                        );
                        break;
                    }
                    Some(Err(e)) => {
                        println!(
                            "[HEARTBEAT] STEP 4 ❌ ws error: {} — socket lived {} s",
                            e,
                            connected_at.elapsed().as_secs()
                        );
                        break;
                    }
                }
            }
        }
    }

    println!(
        "[HEARTBEAT] STEP 5 ── loop exited after {} tick(s), total lifetime {} s (user = {:?})",
        tick_count,
        connected_at.elapsed().as_secs(),
        session.user
    );
    println!("════════════════════════════════════════════════════════════");
}

fn mask(token: &str) -> String {
    if token.len() <= 8 {
        return "*".repeat(token.len());
    }
    format!("{}…{}", &token[..4], &token[token.len() - 4..])
}

        .to_string();

        header::CONTENT_DISPOSITION,
        format!("inline; filename=\"{}\"", name).parse().unwrap(),
    );

    println!("[get_resource] SUCCESS: returning {} bytes", bytes.len());
    (StatusCode::OK, headers, bytes).into_response()
}

        teacher.id
    );
    let student: UserInfo = match db()
        .users()
        .find_one(doc! { "id": &attempt.user })
        .await
    {
        Ok(Some(s)) => s,
        Ok(None) => {
            println!(
                "[GRADE_ATTEMPT] STEP 6 ❌ attempt's user {:?} not found",
                attempt.user
            );
            return "FAIL:Student not found".to_string();
        }
        Err(e) => {
            println!("[GRADE_ATTEMPT] STEP 6 ❌ DB error: {}", e);
            return "FAIL:Database error".to_string();
        }
    };

    if student.teacher != teacher.id {
        // Fall back to a name comparison for older docs that stored the
        // teacher's name instead of their id.
        let matches_by_name = student.teacher == teacher.name;
        if !matches_by_name {
            println!(
                "[GRADE_ATTEMPT] STEP 6 ❌ student {:?} belongs to {:?}, not to teacher {:?}",
                student.id, student.teacher, teacher.id
            );
            return "FAIL:Not authorized to grade this attempt".to_string();
        }
        println!("[GRADE_ATTEMPT] STEP 6 ✅ ownership matched by teacher name (legacy doc)");
    } else {
        println!("[GRADE_ATTEMPT] STEP 6 ✅ student belongs to this teacher");
    }

    // ── 7) WRITE THE GRADE ──────────────────────────────────────────
    println!("[GRADE_ATTEMPT] STEP 7 ── writing grade = {:?}", grade);
    match db()
        .attempts()
        .update_one(
            doc! { "id": &attempt.id },
            doc! { "$set": { "grade": grade } },
        )
        .await
    {
        Ok(res) if res.matched_count > 0 => {
            println!(
                "[GRADE_ATTEMPT] STEP 7 ✅ grade written ({} bytes)",
                grade.len()

use crate::identity::identities::Session;
use axum::Json;

#[derive(Debug, Deserialize)]
pub struct SessionRequest {
    pub token: String,
}

/// POST /session/check
/// Body: { "token": "..." }
/// Returns:
///   "OK:<new_token>" → valid, refreshed (rotated token + slid 1 hour)
///   "EXPIRED"        → past ends_at
///   "NOT_FOUND"      → no session
///   "ERROR:..."      → db/parse failure
pub async fn check_session(Json(body): Json<SessionRequest>) -> String {
    let session: Session = match db()
        .sessions()
        .find_one(doc! { "token": &body.token })
        .await
    {
        Ok(Some(s)) => s,
        Ok(None) => return "NOT_FOUND".to_string(),
        Err(e) => {
            println!("CHECK SESSION DB ERROR: {}", e);
            return format!("ERROR:{}", e);
        }
    };

    let now = chrono::Utc::now();
    let ends_at = match chrono::DateTime::parse_from_rfc3339(&session.ends_at) {
        Ok(dt) => dt.with_timezone(&chrono::Utc),
        Err(e) => {
            println!("CHECK SESSION PARSE ERROR: {}", e);
            return format!("ERROR:{}", e);
        }
    };

    if now >= ends_at {
        return "EXPIRED".to_string();
    }

    let new_token = CryptoService::random_string(32);
    let new_end = (now + chrono::Duration::hours(1)).to_rfc3339();

    if let Err(e) = db()
        .sessions()
        .update_one(
            doc! { "token": &body.token },
            doc! { "$set": {
                "token":      &new_token,
                "updated_at": now.to_rfc3339(),
                "ends_at":    new_end,
            }},
        )
        .await
    {
        println!("CHECK SESSION REFRESH ERROR: {}", e);
        return format!("ERROR:{}", e);
    }

    format!("OK:{}", new_token)
}

/// POST /session/refresh
/// Body: { "token": "..." }
/// Returns:
///   "OK:<new_token>" → exists, still valid, rotated + extended 1 hour
///   "EXPIRED"        → past ends_at (deleted, not revived)
///   "NOT_FOUND"      → no session
///   "ERROR:..."      → db/parse failure
pub async fn refresh_session(Json(body): Json<SessionRequest>) -> String {
    let session: Session = match db()
        .sessions()
        .find_one(doc! { "token": &body.token })
        .await
    {
        Ok(Some(s)) => s,
        Ok(None) => return "NOT_FOUND".to_string(),
        Err(e) => {
            println!("REFRESH SESSION DB ERROR: {}", e);
            return format!("ERROR:{}", e);
        }
    };

    let now = chrono::Utc::now();
    let ends_at = match chrono::DateTime::parse_from_rfc3339(&session.ends_at) {
        Ok(dt) => dt.with_timezone(&chrono::Utc),
        Err(e) => {
            println!("REFRESH SESSION PARSE ERROR: {}", e);
            return format!("ERROR:{}", e);
        }
    };

    if now >= ends_at {
        let _ = db()
            .sessions()
            .delete_one(doc! { "token": &body.token })
            .await;
        return "EXPIRED".to_string();
    }

    let new_token = CryptoService::random_string(32);
    let new_end = (now + chrono::Duration::hours(1)).to_rfc3339();

    if let Err(e) = db()
        .sessions()
        .update_one(
            doc! { "token": &body.token },
            doc! { "$set": {
                "token":      &new_token,
                "updated_at": now.to_rfc3339(),
                "ends_at":    new_end,
            }},
        )
        .await
    {
        println!("REFRESH SESSION UPDATE ERROR: {}", e);
        return format!("ERROR:{}", e);
    }

    format!("OK:{}", new_token)
}

use axum::{response::IntoResponse, Json};

use crate::components::dbfactory::db;

use crate::identity::identities::{RequestGroupAction, Session, Teacher, UserInfo};

#[derive(Debug, Serialize)]
pub struct StudentPassword {
    code: String,
    password: String,
}

pub async fn rotate_passwords(
    Json(body): Json<RequestGroupAction>,
) -> impl IntoResponse {
    // ── 1) SESSION EXISTS ────────────────────────────────────────────
    let session: Session = match db()
        .sessions()
        .find_one(doc! { "token": &body.token })
        .await
    {
        Ok(Some(s)) => s,
        Ok(None) => return "FAIL:Invalid session".to_string(),
        Err(e) => {
            println!("ROTATE PASSWORDS SESSION ERROR: {}", e);
            return "FAIL:Database error".to_string();
        }
    };

    // ── 2) SESSION NOT EXPIRED ───────────────────────────────────────
    let now = chrono::Utc::now();
    let ends_at = match chrono::DateTime::parse_from_rfc3339(&session.ends_at) {
        Ok(dt) => dt.with_timezone(&chrono::Utc),
        Err(e) => {
            println!("ROTATE PASSWORDS PARSE ERROR: {}", e);
            return "FAIL:Invalid session format".to_string();
        }
    };
    if now >= ends_at {
        return "FAIL:Session expired".to_string();
    }

    // ── 3) MUST BE A TEACHER ─────────────────────────────────────────
    let teacher: Teacher = match db()
        .teachers()
        .find_one(doc! { "id": &session.user })
        .await
    {
        Ok(Some(t)) => t,
        Ok(None) => return "FAIL:Only teachers can rotate passwords".to_string(),
        Err(e) => {
            println!("ROTATE PASSWORDS TEACHER ERROR: {}", e);
            return "FAIL:Database error".to_string();
        }
    };

    // ── 4) FETCH STUDENTS IN THAT GROUP FOR THIS TEACHER ────────────
    let mut cursor = match db()
        .users()
        .find(doc! { "group": &body.group, "teacher": &teacher.id })
        .await
    {
        Ok(c) => c,
        Err(e) => {
            println!("ROTATE PASSWORDS FIND ERROR: {}", e);
            return "FAIL:Database error".to_string();
        }
    };

    let mut students: Vec<UserInfo> = Vec::new();
    while cursor.advance().await.unwrap_or(false) {
        if let Ok(u) = cursor.deserialize_current() {
            students.push(u);
        }
    }

    // ── 5) SORT BY NUMERIC SUFFIX IN CODE ───────────────────────────
    students.sort_by_key(|u| {
        u.code
            .chars()
            .skip_while(|c| !c.is_ascii_digit())
            .collect::<String>()
            .parse::<u32>()
            .unwrap_or(0)
    });

    // ── 6) ROTATE + UPDATE EACH STUDENT ────────────────────────────
    let mut rows: Vec<StudentPassword> = Vec::with_capacity(students.len());

    for student in students {
        let new_password = CryptoService::random_string(4).to_uppercase();

        match db()
            .users()
            .update_one(
                doc! { "id": &student.id },
                doc! { "$set": { "password": &new_password } },
            )
            .await
        {
            Ok(res) if res.matched_count > 0 => {
                rows.push(StudentPassword {
                    code: student.code,
                    password: new_password,
                });
            }
            Ok(_) => {
                println!("ROTATE PASSWORDS: student {} vanished mid-update", student.code);
            }
            Err(e) => {
                println!("ROTATE PASSWORDS UPDATE ERROR ({}): {}", student.code, e);
            }
        }
    }

    // ── 7) RETURN AS JSON ───────────────────────────────────────────
    match serde_json::to_string(&rows) {
        Ok(json) => format!("OK:{}", json),
        Err(e) => {
            println!("ROTATE PASSWORDS SERIALIZE ERROR: {}", e);
            "FAIL:Failed to serialize result".to_string()
        }
    }
}

use axum::{response::IntoResponse, Json};
use mongodb::bson::doc;
use serde::Deserialize;

use crate::components::dbfactory::db;
use crate::components::encrypt::CryptoService;
use crate::identity::identities::{
    AssigmentAttempt, AssigmentAttemptResource, RawAssigment, ResourceAccess, Session, UserInfo,
};

const ATTEMPT_WINDOW_MINUTES: u32 = 15;

pub async fn start_attempt(Json(body): Json<ResourceAccess>) -> impl IntoResponse {
    println!("════════════════════════════════════════════════════════════");
    println!("[START_ATTEMPT] ── INCOMING REQUEST ───────────────────────");

    // ── 1) SESSION EXISTS ────────────────────────────────────────────
    println!("[START_ATTEMPT] STEP 1 ── looking up session");
    let session: Session = match db()
        .sessions()
        .find_one(doc! { "token": &body.token })
        .await
    {
        Ok(Some(s)) => {
            println!("[START_ATTEMPT] STEP 1 ✅ session found for user = {:?}", s.user);
            s
        }
        Ok(None) => {
            println!("[START_ATTEMPT] STEP 1 ❌ no session for that token");
            return "FAIL:Invalid session".to_string();
        }
        Err(e) => {
            println!("[START_ATTEMPT] STEP 1 ❌ DB error: {}", e);
            return "FAIL:Database error".to_string();
        }
    };

    // ── 2) SESSION NOT EXPIRED ───────────────────────────────────────
    println!("[START_ATTEMPT] STEP 2 ── checking expiry");
    let now = chrono::Utc::now();
    let ends_at = match chrono::DateTime::parse_from_rfc3339(&session.ends_at) {
        Ok(dt) => dt.with_timezone(&chrono::Utc),
        Err(e) => {
            println!("[START_ATTEMPT] STEP 2 ❌ parse error: {}", e);
            return "FAIL:Invalid session format".to_string();
        }
    };
    if now >= ends_at {
        println!("[START_ATTEMPT] STEP 2 ❌ session expired");
        return "FAIL:Session expired".to_string();
    }
    println!("[START_ATTEMPT] STEP 2 ✅ session not expired");

    // ── 3) MUST BE A STUDENT ─────────────────────────────────────────
    println!("[START_ATTEMPT] STEP 3 ── verifying student = {:?}", session.user);
    let student: UserInfo = match db()
        .users()
        .find_one(doc! { "id": &session.user })
        .await
    {
        Ok(Some(u)) => {
            println!(
                "[START_ATTEMPT] STEP 3 ✅ student found, teacher = {:?}",
                u.teacher
            );
            u
        }
        Ok(None) => {
            println!("[START_ATTEMPT] STEP 3 ❌ not a student");
            return "FAIL:Only students can start attempts".to_string();
        }
        Err(e) => {
            println!("[START_ATTEMPT] STEP 3 ❌ DB error: {}", e);
            return "FAIL:Database error".to_string();
        }
    };

    // ── 4) RESOLVE THE STUDENT'S TEACHER ID ─────────────────────────
    let teacher_id: String = {
        let by_id = db()
            .teachers()
            .find_one(doc! { "id": &student.teacher })
            .await
            .ok()
            .flatten();

        if let Some(t) = by_id {
            t.id
        } else {
            let by_name = db()
                .teachers()
                .find_one(doc! { "name": &student.teacher })
                .await
                .ok()
                .flatten();
            match by_name {
                Some(t) => t.id,
                None => {
                    println!(
                        "[START_ATTEMPT] STEP 4 ❌ student's teacher {:?} not found",
                        student.teacher
                    );
                    return "FAIL:Teacher not found".to_string();
                }
            }
        }
    };
    println!("[START_ATTEMPT] STEP 4 ✅ teacher resolved = {:?}", teacher_id);

    // ── 5) FIND THE ASSIGNMENT ──────────────────────────────────────
    println!("[START_ATTEMPT] STEP 5 ── fetching assignment for teacher = {:?}", teacher_id);
    let assignment: RawAssigment = match db()
        .raw_assigments()
        .find_one(doc! { "teacher": &teacher_id })
        .await
    {
        Ok(Some(a)) => {
            println!(
                "[START_ATTEMPT] STEP 5 ✅ assignment id = {:?}, label = {:?}, type = {:?}",
                a.id, a.label, a.assigment_type
            );
            a
        }
        Ok(None) => {
            println!("[START_ATTEMPT] STEP 5 ❌ no assignment for this teacher");
            return "FAIL:No assignment found".to_string();
        }
        Err(e) => {
            println!("[START_ATTEMPT] STEP 5 ❌ DB error: {}", e);
            return "FAIL:Database error".to_string();
        }
    };

    // ── 6) DETERMINE NEXT ATTEMPT NUMBER ────────────────────────────
    println!(
        "[START_ATTEMPT] STEP 6 ── counting previous attempts by user = {:?} for assignment = {:?}",
        session.user, assignment.id
    );
    let previous_count = match db()
        .attempts()
        .count_documents(doc! {
            "user":         &session.user,
            "assigment_id": &assignment.id,
        })
        .await
    {
        Ok(c) => {
            println!("[START_ATTEMPT] STEP 6   previous attempts = {}", c);
            c as u32
        }
        Err(e) => {
            println!("[START_ATTEMPT] STEP 6 ❌ count error: {}", e);
            return "FAIL:Database error".to_string();
        }
    };

    let next_attempt = previous_count + 1;
    println!("[START_ATTEMPT] STEP 6 ✅ this attempt = {}", next_attempt);

    // ── 7) ENFORCE max_attempts ─────────────────────────────────────
    if next_attempt > assignment.max_attempts {
        println!(
            "[START_ATTEMPT] STEP 7 ❌ max attempts reached ({}/{})",
            previous_count, assignment.max_attempts
        );
        return format!(
            "FAIL:Maximum attempts reached ({}/{})",
            previous_count, assignment.max_attempts
        );
    }
    println!(
        "[START_ATTEMPT] STEP 7 ✅ attempt {} of {}",
        next_attempt, assignment.max_attempts
    );

    // ── 8) CREATE THE ATTEMPT RECORD ────────────────────────────────
    // Snapshot the assignment's identity (id, label, type) onto the
    // attempt itself so the log survives the assignment being wiped
    // by a future update_assigment call.
    println!("[START_ATTEMPT] STEP 8 ── creating attempt record");
    let attempt_id = CryptoService::random_string(32);
    let attempt = AssigmentAttempt {
        id: attempt_id.clone(),
        user: session.user.clone(),
        attempt: next_attempt,
        assigment_id: assignment.id.clone(),
        assigment_label: assignment.label.clone(),
        assigment_type: assignment.assigment_type.clone(),
        max_attempts: assignment.max_attempts,
        grade: String::new(),
        window: ATTEMPT_WINDOW_MINUTES,
        json: String::new(),
    };

    if let Err(e) = db().attempts().insert_one(&attempt).await {
        println!("[START_ATTEMPT] STEP 8 ❌ insert error: {}", e);
        return "FAIL:Database error".to_string();
    }
    println!(
        "[START_ATTEMPT] STEP 8 ✅ attempt id = {:?} (label = {:?}, type = {:?})",
        attempt_id, attempt.assigment_label, attempt.assigment_type
    );

    // ── 9) ISSUE A SHORT-LIVED "attempt" SESSION ────────────────────
    println!("[START_ATTEMPT] STEP 9 ── issuing attempt session");
    let attempt_token = CryptoService::random_string(32);
    let attempt_session = Session {
        id: attempt_id.clone(),
        token: attempt_token.clone(),
        user: session.user,
        created_at: now.to_rfc3339(),
        updated_at: now.to_rfc3339(),
        ends_at: (now + chrono::Duration::minutes(ATTEMPT_WINDOW_MINUTES as i64))
            .to_rfc3339(),
    };

    if let Err(e) = db().sessions().insert_one(&attempt_session).await {
        println!("[START_ATTEMPT] STEP 9 ❌ insert error: {}", e);
        return "FAIL:Database error".to_string();
    }
    println!(
        "[START_ATTEMPT] STEP 9 ✅ attempt session issued (expires in {} min)",
        ATTEMPT_WINDOW_MINUTES
    );

    // ── 10) RETURN JSON + TOKEN ─────────────────────────────────────
    let response = AssigmentAttemptResource {
        json: assignment.json,
        token: attempt_token,
    };

    match serde_json::to_string(&response) {
        Ok(json) => {
            println!("[START_ATTEMPT] STEP 10 ✅ ── DONE ── {} bytes", json.len());
            println!("════════════════════════════════════════════════════════════");
            format!("OK:{}", json)
        }
        Err(e) => {
            println!("[START_ATTEMPT] STEP 10 ❌ serialize error: {}", e);
            println!("════════════════════════════════════════════════════════════");
            "FAIL:Failed to serialize result".to_string()
        }
    }
}

async fn check_daily_quota(user: &str, now: &chrono::DateTime<chrono::Utc>) -> Result<(), String> {
    println!("[QUOTA] checking daily quota for user = {:?}", user);

    let last = match db()
        .files()
        .find_one(doc! { "uploaded_by": user })
        .sort(doc! { "uploaded_date": -1 })
        .await
    {
        Ok(Some(f)) => {
            println!(
                "[QUOTA] found last upload = {:?} at {:?}",
                f.name, f.uploaded_date
            );
            Some(f)
        }
        Ok(None) => {
            println!("[QUOTA] no previous uploads for this user");
            None
        }
        Err(e) => {
            println!("[QUOTA] ❌ find error: {}", e);
            return Err("FAIL:Database error".to_string());
        }
    };

    let start_of_today = now.date_naive().and_hms_opt(0, 0, 0).unwrap().and_utc();
    println!("[QUOTA] start_of_today = {}", start_of_today);

    if let Some(ref last_file) = last {
        if let Ok(last_time) = chrono::DateTime::parse_from_rfc3339(&last_file.uploaded_date) {
            if last_time.with_timezone(&chrono::Utc) < start_of_today {
                println!("[QUOTA] last upload was before today → quota reset, allow");
                return Ok(());
            }
        }
    } else {
        println!("[QUOTA] no history → allow");
        return Ok(());
    }

    let start_str = start_of_today.to_rfc3339();
    let count = match db()
        .files()
        .count_documents(doc! {
            "uploaded_by": user,
            "uploaded_date": { "$gte": &start_str },
        })
        .await
    {
        Ok(c) => {
            println!("[QUOTA] uploads today = {}", c);
            c
        }
        Err(e) => {
            println!("[QUOTA] ❌ count error: {}", e);
            return Err("FAIL:Database error".to_string());
        }
    };

    let is_teacher = db()
        .teachers()
        .find_one(doc! { "id": user })
        .await
        .ok()
        .flatten()
        .is_some();

    let limit = if is_teacher {
        println!("[QUOTA] user is a teacher → limit = {}", TEACHER_DAILY_UPLOAD_LIMIT);
        TEACHER_DAILY_UPLOAD_LIMIT
    } else {
        println!("[QUOTA] student limit = {}", STUDENT_DAILY_UPLOAD_LIMIT);
        STUDENT_DAILY_UPLOAD_LIMIT
    };

    if count as i64 >= limit {
        println!("[QUOTA] ❌ limit reached ({}/{})", count, limit);
        return Err(format!(
            "FAIL:Daily upload limit reached ({}/{})",
            count, limit
        ));
    }

    println!("[QUOTA] ✅ under limit ({}/{})", count, limit);
    Ok(())
}
