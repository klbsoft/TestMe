export default function AudioFile() {
    const {isPlaying, audio_file } = useAssignmentState();
    const update = useUpdateAssignment();

    // one persistent audio element for this component's lifetime
    const audioRef = useRef<HTMLAudioElement | null>(null);

    // ---- 1. create the element once, attach the ended handler ----
    useEffect(() => {
        const audio = new Audio();
        audio.preload = 'auto';

        const handleEnded = () => {
            // reflect reality back into state
            update((prev)=>{
                // console.log(prev);
                if(prev.unitIndex === 0){
                        const index = prev.unitIndex+1; 
                        // const assigment = prev.session.assignment as SpeakingAssignment;
                        // const units = assigment.units as SpeakingUnit[];
                        // const content = units[index].content; 
                          return {
                        unitIndex: index,
                        isPlaying: false,
                        // audio_file: content,
                        // isPlaying: true,
                        // isTicking: false,
                        // message_note: "",
                    }
                }
                if(prev.unitIndex === 6){
                        const index = prev.unitIndex+1; 
                        const assigment = prev.session.assignment as SpeakingAssignment;
                        const units = assigment.units as SpeakingUnit[];
                        const content = units[index].content; 
                        // console.log({content,index})
                    return {
                        unitIndex: index,
                        audio_file: content,
                        isPlaying: true,
                        isTicking: false,
                        message_note: "",
                    }
                }
                  
                return {
                        isPlaying: false,
                        isTicking: prev.unitIndex > 6 ? true:false,
                        message_note: "",
                }
            });

        };
        audio.addEventListener('ended', handleEnded);

        audioRef.current = audio;

        return () => {
            audio.removeEventListener('ended', handleEnded);
            audio.pause();
            audio.src = '';           // release buffered data
            audioRef.current = null;
        };
        // update is stable from your store; if not, include it
        // eslint-disable-next-line react-hooks/exhaustive-deps
    }, []);

    // ---- 2. whenever audio_file changes, swap src and reload ----
    useEffect(() => {
        // console.log("changed");
        const audio = audioRef.current;
        // console.log({current:audio.src}); 
        // console.log({new:audio_file});
        if (!audio) return;
         
        // only touch src if it actually changed
        if (audio_file && audio.src !== audio_file) {
            audio.src = audio_file;
            audio.load();             // forces reload of the new file
        }
    }, [audio_file]);

    // ---- 3. whenever isPlaying changes, play / pause ----
useEffect(() => {

  const timeoutId = setTimeout(() => {
        const audio = audioRef.current;
        // console.log(audio.src);
        if (!audio) return;
            if (isPlaying) {
            if (!audio.src) return;
            audio.play().catch((err) => {
                console.error('play failed', err);
                update({ isPlaying: false });
            });
            } else {
            audio.pause();
            }
        }, 500);

  return () => clearTimeout(timeoutId);
}, [isPlaying, audio_file]);

    return <div />;
}

import { useEffect, useState, type ReactNode } from "react";
import CheatDetected from "./CheatDetected";

type Props = {
  children: ReactNode;
  enabled?: boolean;
  onCheatDetected?: (reason: string) => void;
};

export default function CheatDetector({ children, enabled = true, onCheatDetected }: Props) {
  const [cheatReason, setCheatReason] = useState<string | null>(null);

  useEffect(() => {
    if (!enabled) return;   // don't attach listeners when disabled

    const flag = (reason: string) => {
      setCheatReason((prev) => prev ?? reason);
      onCheatDetected?.(reason);
    };

    const handleBlur = () => flag("window-blur");
    const handleFullscreenChange = () => {
      if (!document.fullscreenElement) flag("exited-fullscreen");
    };

    window.addEventListener("blur", handleBlur);
    document.addEventListener("fullscreenchange", handleFullscreenChange);

    return () => {
      window.removeEventListener("blur", handleBlur);
      document.removeEventListener("fullscreenchange", handleFullscreenChange);
    };
  }, [enabled, onCheatDetected]);

  if (cheatReason) {
    return (
      <>
        <div style={{ display: "none" }}>{children}</div>
        <div style={{ position: "fixed", inset: 0, zIndex: 9999, backgroundColor: "#fff", overflow: "auto" }}>
          <CheatDetected reason={cheatReason} />
        </div>
      </>
    );
  }

  return <>{children}</>;
}
