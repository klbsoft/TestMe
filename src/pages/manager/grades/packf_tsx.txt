/**
 * POST /assigment/attempts
 *
 * - `token` empty            → toast the error, return the failure string.
 * - network / parse failure  → toast the error, return the failure string.
 * - `FAIL:...` from server   → toast the error, return the failure string.
 * - `OK:[...]` from server   → return the parsed `AttemptRow[]`.
 */
export async function getAttempts(token: string): Promise<AttemptRow[] | string> {
  if (!token) {
    const msg = "FAIL:No session token";
    showToast(msg);
    return msg;
  }

  let text: string;
  try {
    const res = await fetch(`${defaultSetting.api_url}/assigment/attempts`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ token }),
    });
    text = await res.text();
  } catch (e) {
    const msg = `FAIL:Network error: ${String(e)}`;
    showToast(msg);
    return msg;
  }

  if (!text.startsWith("OK:")) {
    showToast(text);
    return text.startsWith("FAIL:") ? text : `FAIL:${text}`;
  }

  try {
    const rows = JSON.parse(text.slice(3)) as AttemptRow[];
    console.log(rows);
    return rows;
  } catch (e) {
    const msg = `FAIL:Invalid attempts payload: ${String(e)}`;
    showToast(msg);
    return msg;
  }
}

/**
 * POST /assigment/attempt/grade
 *
 * Body: { token, attempt, grade }
 *   - `attempt` is the attempt id (the `id` field on AttemptRow)
 *   - `grade` is "0".."4" or "" for ungraded
 *
 * Returns the server's raw response: `OK:...` or `FAIL:...`.
 * Toasts any failure and returns it as a string so the caller can react.
 */
export async function gradeAttempt(
  attemptId: string,
  grade: string
): Promise<string> {
  const token = getToken();
  if (!token) {
    const msg = "FAIL:No session token";
    showToast(msg);
    return msg;
  }

  let text: string;
  try {
    const res = await fetch(
      `${defaultSetting.api_url}/assigment/attempt/grade`,
      {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          token,
          attempt: attemptId,
          grade,
        }),
      }
    );
    text = await res.text();
  } catch (e) {
    const msg = `FAIL:Network error: ${String(e)}`;
    showToast(msg);
    return msg;
  }

  if (!text.startsWith("OK:")) {
    showToast(text);
    return text.startsWith("FAIL:") ? text : `FAIL:${text}`;
  }

  showToast("OK:Grade saved");
  return text;
}

async function fetchResourceBlob(
  resourceUrl: string,
  token: string
): Promise<string | null> {
  try {
    const res = await fetch(resourceUrl, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ token }),
    });
    if (!res.ok) return null;
    const ct = res.headers.get("content-type") ?? "";
    if (ct.startsWith("text/")) return null;
    const blob = await res.blob();
    if (blob.size === 0) return null;
    return URL.createObjectURL(blob);
  } catch {
    return null;
  }
}

async function hydrateUnits(
  units: SpeakingUnit[],
  token: string
): Promise<{ hydrated: SpeakingUnit[]; blobUrls: string[] }> {
  const blobUrls: string[] = [];

  const hydrated = await Promise.all(
    units.map(async (unit) => {
      const next: SpeakingUnit = { ...unit };

      if (next.content) {
        const blob = await fetchResourceBlob(next.content, token);
        if (blob) {
          blobUrls.push(blob);
          next.content = blob;
        }
      }
      if (next.answer) {
        const blob = await fetchResourceBlob(next.answer, token);
        if (blob) {
          blobUrls.push(blob);
          next.answer = blob;
        }
      }

      return next;
    })
  );

  return { hydrated, blobUrls };
}

/** The subset of units that UnitsList renders (and therefore grades). */
function gradedUnitsOf(units: SpeakingUnit[]): SpeakingUnit[] {
  const contentUnits = units.filter((u) => !u.id.startsWith("dialog_"));
  const describeUnits = contentUnits
    .filter((u) => u.phase === "describe_image")
    .slice(0, 4);
  const repeatUnits = contentUnits
    .filter((u) => u.phase === "repeat_audio")
    .slice(0, 4);
  return describeUnits.length + repeatUnits.length > 0
    ? [...describeUnits, ...repeatUnits]
    : contentUnits.slice(0, 8);
}
